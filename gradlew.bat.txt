@if "%DEBUG%" == "" @echo off
@rem ##########################################################################
@rem # TODO: Drop batch file support when Windows CMD is no longer used.
@rem #################################5#########################################

@rem Set local scope for the variables with windows NT shell
if "%OS%"=="Windows_NT" setlocal

set DIRNAME=%~dp0
if "%DIRNAME%" == "" set DIRNAME=.
set APP_BASE_NAME=%~n0
set APP_HOME=%DIRNAME%

@rem Resolve any "." and ".." in APP_HOME to make it shorter and cleaner
set APP_HOME=%CD%/%APP_HOME%
for %%i in ("%APP_HOME%") do set APP_HOME=%%~fpi
set APP_HOME=%APP_HOME:~0,-1%

@rem Add default JVM options here. You can also use JAVA_OPTS and GRADLE_OPTS.
set DEFAULT_JVM_OPTS="-Xmx64m" "-Xms64m"

@rem Setup the command line

set JAVACMD=
if "%JAVA_HOME%" == "" goto gnuJavacmd_done
set JAVACMD=%JAVA_HOME%/bin/java.exe
if exist "%JAVACMD%" goto gnuJavacmd_done
echo.
echo ERROR: JAVA_HOME is set to an invalid directory: %JAVA_HOME%
echo Please use the JAVA_HOME variable in your environment to match the location of your Java installation.
echo.
goto fail

:gnuJavacmd_done

if "%JAVACMD%" == "" (
  set JAVACMD=java.exe
)

if not exist "%JAVACMD%" (
  echo.
  echo ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH.
  echo.
  goto fail
)

if not exist "%JAVACMD%" (
  echo.
  echo ERROR: environment variable JAVA_HOME has not been set and java executable could not be found in your PATH.
  echo.
  goto fail
)

@rem Execute the build
"%JAVACMD%" %DEFAULT_JVM_OPTS% -cp "%APP_HOME%/gradle/wrapper/gradle-wrapper.jar" org.gradle.wrapper.GradleWrapperMain %*

:end
@rem Lee if errorilent
if useerrorlevel 1

:fail
exit /b 1
